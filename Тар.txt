        function initTheme() {
            if (localStorage.theme === 'dark' || (!('theme' in localStorage) && window.matchMedia('(prefers-color-scheme: dark)').matches)) {
                document.documentElement.classList.add('dark');
            } else {
                document.documentElement.classList.remove('dark');
            }
        }

        function toggleTheme() {
            if (document.documentElement.classList.contains('dark')) {
                document.documentElement.classList.remove('dark');
                localStorage.theme = 'light';
            } else {
                document.documentElement.classList.add('dark');
                localStorage.theme = 'dark';
            }
        }

        function goHome() {
            currentState.view = 'home';
            renderApp();
        }

        function shuffleArray(array) {
            const arr = [...array];
            for (let i = arr.length - 1; i > 0; i--) {
                const j = Math.floor(Math.random() * (i + 1));
                [arr[i], arr[j]] = [arr[j], arr[i]];
            }
            return arr;
        }

        function startQuiz() {
            const processedQuestions = QUIZ_QUESTIONS.map(q => {
                const questionText = q[0];
                const choices = q[1];
                const correctIdx = q[2];
                const explanation = q[3];

                const mappedChoices = choices.map((c, idx) => ({ text: c, isCorrect: idx === correctIdx }));
                const shuffledChoices = shuffleArray(mappedChoices);

                return {
                    questionText,
                    choices: shuffledChoices,
                    explanation
                };
            });

            currentState.view = 'quiz';
            currentState.questions = shuffledQuestions;
            currentState.currentIndex = 0;
            currentState.userAnswers = [];

            renderApp();
        }

        function handleAnswer(choiceIdx) {
            if (currentState.userAnswers[currentState.currentIndex] !== undefined) return;

            const currentQ = currentState.questions[currentState.currentIndex];
            const isCorrect = currentQ.choices[choiceIdx].isCorrect;
            currentState.userAnswers[currentState.currentIndex] = {
                choiceIndex: choiceIdx,
                isCorrect: isCorrect
            };

            const app = document.getElementById('app');
            if (app) renderQuiz(app);
        }

        function nextQuestion() {
            if (currentState.currentIndex < currentState.questions.length - 1) {
                currentState.currentIndex++;
                const app = document.getElementById('app');
                if (app) renderQuiz(app);
            } else {
                finishQuiz();
            }
        }

        function finishQuiz() {
            const score = currentState.userAnswers.filter(a => a && a.isCorrect).length;
            const prev = currentState.stats.best || { attempts: 0, maxScore: 0 };
            
            currentState.stats.best = {
                attempts: prev.attempts + 1,
                maxScore: Math.max(prev.maxScore, score),
                lastScore: score,
                date: new Date().toLocaleDateString()
            };
            saveStats();

            currentState.view = 'result';
            renderApp();
        }

        function renderApp() {
            const app = document.getElementById('app');
            if (!app) return;
            app.innerHTML = '';

            if (currentState.view === 'home') {
                renderHome(app);
            } else if (currentState.view === 'quiz') {
                renderQuiz(app);
            } else if (currentState.view === 'result') {
                renderResult(app);
            }
        }

        function toggleSearchModal() {
            const modal = document.getElementById('search-modal');
            if (modal) {
                modal.classList.toggle('hidden');
                if (!modal.classList.contains('hidden')) {
                    document.getElementById('search-input').focus();
                }
            }
        }

        function searchQuestions(query) {
            const resultsContainer = document.getElementById('search-results');
            if (!resultsContainer) return;

            if (!query.trim()) {
                resultsContainer.innerHTML = `<p class="text-center text-slate-400 text-xs sm:text-sm py-8">Кілт сөзді енгізіңіз (мысалы: "Ботай", "Палеолит", "Микролит")</p>`;
                return;
            }

            const term = query.toLowerCase();
            const matches = QUIZ_QUESTIONS.filter(q => q[0].toLowerCase().includes(term) || q[3].toLowerCase().includes(term));

            if (matches.length === 0) {
                resultsContainer.innerHTML = `<p class="text-center text-slate-400 text-xs sm:text-sm py-8">Сұрақ табылған жоқ</p>`;
                return;
            }

            resultsContainer.innerHTML = matches.map(q => `
                <div class="p-3 bg-slate-50 dark:bg-slate-800/60 rounded-xl border border-slate-200 dark:border-slate-700/60">
                    <p class="font-bold text-xs sm:text-sm text-slate-900 dark:text-slate-100 mb-1">${q[0]}</p>
                    <p class="text-xs text-emerald-600 dark:text-emerald-400 font-semibold mb-1">Дұрыс жауап: ${q[1][q[2]]}</p>
                    <p class="text-[11px] sm:text-xs text-slate-500 dark:text-slate-400">${q.explanation || q[3]}</p>
                </div>
            `).join('');
        }
